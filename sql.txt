SELECT
    c.login AS "Логин курьера",
    COUNT(o.id) AS "Количество заказов в доставке"
FROM
    "Couriers" c
LEFT JOIN
    "Orders" o ON o."courierId" = c.id AND o."inDelivery" = true
GROUP BY
    c.login
ORDER BY
    c.login


SELECT
    "track",
    CASE
        WHEN "finished" = true THEN 2
        WHEN "cancelled" = true THEN -1
        WHEN "inDelivery" = true THEN 1
        ELSE 0
    END AS "status"
FROM
    "Orders"
